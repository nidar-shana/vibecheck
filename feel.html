#include using namespace std; const int MOD = 998244353; const int N = 5005; int add(int x, int y) { x += y; if (x >= MOD) x -= MOD; if (x < 0) x += MOD; return x; } int mul(int x, int y) { return 1LL * x * y % MOD; } int sub(int x, int y) { return add(x, -y); } int binpow(int x, int y) { int z = 1; while (y) { if (y & 1) z = mul(z, x); x = mul(x, x); y >>= 1; } return z; } int fact[N], inv[N]; void precalc() { fact[0] = 1; used[x] = true; b.push_back({x, len}); cutoff_pos[x] = i; int c = mul(sign, mul(len, fact[cnt])); int cG = mul(sign, mul(len, fact[cnt - 1])); --cnt; sign = sub(0, sign); pref.push_back(add(pref.back(), c)); pref_G.push_back(add(pref_G.back(), cG)); } a = b; } int getF() { return pref.back(); } int getG(int x) { if (cutoff_pos[x] == -1) return pref_G.back(); return pref_G[cutoff_pos[x]]; } void append(int x) { if (a.empty() || a[0].first != x) a.insert(a.begin(), {x, 0}); a[0].second++; rebuild(); } }; void solve() { int n; cin >> n; vector a(n - 1); for (int i = 0; i < n - 1; i++) { cin >> a[i]; --a[i]; } Blocks b(n); vector> ans(n - 1, vector(n, 0)); vector> G(n - 1, vector(n, 0));
